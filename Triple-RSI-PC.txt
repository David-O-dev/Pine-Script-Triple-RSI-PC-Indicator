//@version=6
indicator("MY RSI")

rsi1Length = input.int(defval = 7, title = "RSI 1 Line", group = "RSI")
rsi2Length = input.int(defval = 9, title = "RSI 2 Line", group = "RSI")
rsi3Length = input.int(defval = 12, title = "RSI 3 Line", group = "RSI")
RSI1 = ta.rsi(close, rsi1Length)
RSI2 = ta.rsi(close, rsi2Length)
RSI3 = ta.rsi(close, rsi3Length)

plot(RSI1, color = color.green, linewidth = 2)
plot(RSI2, color = color.orange, linewidth = 2)
plot(RSI3, color = color.red, linewidth = 2)
top = plot(55, color = color.green)
bottom = plot(45, color = color.red)



//PC for RSI 1
UsePC1 = input.bool(defval = true, title = "PC 1 Switch", group = "PC Switches")
PClength1 = input(9, group = "PC 1")
offsetlength1 = input(0, group = "PC 1")
PCColor1 = input.color(defval = color.black, title = "Price Channel 1 Color", group = "PC 1")

highprice1 = ta.highest(RSI1, PClength1)
lowprice1 = ta.lowest(RSI1, PClength1)
plot(UsePC1 ? highprice1 : na, color=PCColor1, title="Highprice Line 1", offset=offsetlength1, linewidth = 4)
plot(UsePC1 ? lowprice1 : na, color=PCColor1, title="Lowprice Line 1", offset=offsetlength1, linewidth = 4)
